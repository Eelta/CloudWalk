cmake_minimum_required(VERSION 3.24)
project(CloudWalk VERSION 0.5.7 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CLOUDWALK_COMMONLIB_SOURCE "" CACHE PATH "CommonLibSSE-NG 12.2.0 source tree")
set(CLOUDWALK_DEPS "" CACHE PATH "vcpkg dependency prefix")
if(NOT EXISTS "${CLOUDWALK_COMMONLIB_SOURCE}/include/SKSE/Version.h")
  message(FATAL_ERROR "Set CLOUDWALK_COMMONLIB_SOURCE to CommonLibSSE-NG 12.2.0 source, as pinned in dependency-lock.json.")
endif()
file(READ "${CLOUDWALK_COMMONLIB_SOURCE}/include/SKSE/Version.h" CLOUDWALK_SDK_VERSIONS)
if(NOT CLOUDWALK_SDK_VERSIONS MATCHES "RUNTIME_SSE_1_7_104")
  message(FATAL_ERROR "This SDK predates Skyrim 1.7.104. Use the source pinned in dependency-lock.json.")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CLOUDWALK_DEPS}")
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_PREBUILT OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_XBYAK OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLOUDWALK_COMMONLIB_SOURCE}" commonlib)
add_library(CloudWalk SHARED src/ExternalPlugin.cpp src/MediaClient.cpp)
set_source_files_properties(src/MediaClient.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_include_directories(CloudWalk PRIVATE include)
target_compile_definitions(CloudWalk PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(CloudWalk PRIVATE /EHsc /utf-8 /permissive- /Zc:preprocessor /Zi)
target_precompile_headers(CloudWalk PRIVATE include/PCH.h)
target_link_libraries(CloudWalk PRIVATE CommonLibSSE::CommonLibSSE xinput9_1_0 windowsapp ole32 shell32 user32 advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi version)
target_link_options(CloudWalk PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:REF /OPT:ICF)
set_target_properties(CloudWalk PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/release" PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/release")
set_target_properties(CloudWalk PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/release" PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/release")

option(CLOUDWALK_BUILD_TESTS "Build local diagnostics and regression programs" OFF)
if(CLOUDWALK_BUILD_TESTS)
add_executable(media_probe tests/media_probe.cpp src/MediaClient.cpp)
target_include_directories(media_probe PRIVATE include)
target_compile_definitions(media_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(media_probe PRIVATE /EHsc /utf-8)
target_link_libraries(media_probe PRIVATE windowsapp ole32 shell32 user32)
add_executable(background_window_tests tests/background_window_tests.cpp)
target_include_directories(background_window_tests PRIVATE include)
target_compile_definitions(background_window_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(background_window_tests PRIVATE /EHsc /utf-8)
target_link_libraries(background_window_tests PRIVATE user32)
set_target_properties(background_window_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin")
add_executable(player_regression_tests tests/player_regression_tests.cpp)
target_include_directories(player_regression_tests PRIVATE include)
target_compile_options(player_regression_tests PRIVATE /EHsc /utf-8)
enable_testing()
add_test(NAME player_input_and_original_music_restore COMMAND player_regression_tests)
add_test(NAME background_window_order_and_minimize COMMAND background_window_tests)

set_target_properties(media_probe player_regression_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin")

set_target_properties(media_probe player_regression_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin")

add_executable(runtime_compatibility_tests tests/runtime_compatibility_tests.cpp)
target_include_directories(runtime_compatibility_tests PRIVATE include)
target_compile_definitions(runtime_compatibility_tests PRIVATE ENABLE_COMMONLIBSSE_TESTING=1 NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(runtime_compatibility_tests PRIVATE /EHsc /utf-8 /Zc:preprocessor)
target_precompile_headers(runtime_compatibility_tests PRIVATE include/PCH.h)
target_link_libraries(runtime_compatibility_tests PRIVATE CommonLibSSE::CommonLibSSE)
set_target_properties(runtime_compatibility_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/qa/bin")
add_test(NAME runtime_declaration_and_version_gates COMMAND runtime_compatibility_tests)
endif()
